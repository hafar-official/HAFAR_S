// Fungsi membaca file video berukuran besar (hingga 200 MB) menjadi Base64 secara efisien function convertLargeFileToBase64(file) { return new Promise((resolve, reject) => { const chunkSize = 2 * 1024 * 1024; // 2MB per bagian let offset = 0; let base64Result = ""; const reader = new FileReader(); reader.onload = (e) => { base64Result += e.target.result.split(',')[1] || ""; offset += chunkSize; if (offset < file.size) { const percentage = Math.round((offset / file.size) * 100); document.getElementById('upload-video-status').innerText = `Memproses video: ${percentage}%...`; readNextChunk(); } else { resolve(`data:${file.type};base64,` + base64Result); } }; reader.onerror = (error) => reject(error); function readNextChunk() { const slice = file.slice(offset, offset + chunkSize); reader.readAsDataURL(slice); } readNextChunk(); }); } async function uploadVideo() { const fileInput = document.getElementById('video-input'); const titleInput = document.getElementById('video-title-input').value.trim(); const file = fileInput.files[0]; const status = document.getElementById('upload-video-status'); const uploadBtn = document.getElementById('upload-video-btn'); if (!file) return alert("Pilih file video terlebih dahulu!"); // Batas maksimal diperbarui menjadi 200 MB const maxSizeBytes = 200 * 1024 * 1024; if (file.size > maxSizeBytes) { return alert("Ukuran video terlalu besar! Maksimal 200 MB."); } const videoTitle = titleInput || file.name; status.innerText = "Memulai proses video..."; uploadBtn.disabled = true; try { const base64Data = await convertLargeFileToBase64(file); status.innerText = "Mengirim data ke database..."; await db.collection('videos').add({ title: videoTitle, url: base64Data, createdAt: new Date() }); status.innerText = "✓ Video berhasil di-upload!"; fileInput.value = ""; document.getElementById('video-title-input').value = ""; uploadBtn.disabled = false; loadVideos(); } catch (err) { uploadBtn.disabled = false; status.innerText = "Gagal upload: " + err.message; } }
